================================================================================
PRICING ORCHESTRATOR - COMPLETE IMPLEMENTATION
Cartesian Product of SDE × FDM × RNG Combinations
================================================================================

OVERVIEW
--------
This implementation creates a mediator instance for every combination of:
- Stochastic Differential Equations (SDEs): GBM, CEV, etc.
- Finite Difference Methods (FDMs): Euler, Milstein, Exact, etc.
- Random Number Generators (RNGs): MersenneTwister, BoxMuller, etc.

For example, with 2 SDEs, 3 FDMs, and 2 RNGs, you get 2×3×2 = 12 mediators.

// Structure to hold a complete simulation configuration
struct SimulationInstance
{
	std::unique_ptr<Mediator> mediator;
	std::vector<std::shared_ptr<Pricer>> pricers;

	std::string fdmType;
	std::string rngType;
	int instanceId;
	
	SimulationInstance(int id, const std::string& sde, const std::string& fdm, const std::string& rng)
		: instanceId(id), sdeType(sde), fdmType(fdm), rngType(rng) {}
};

class PricingOrchestrator

private:

	std::vector<SimulationInstance> simulations_;

	// Component factories

	void setupFiniteDifferenceMethods();
	void setupStochasticDifferentialEquations();

public:

	// Allow move operations
	PricingOrchestrator(PricingOrchestrator&&) = default;

	// Main orchestration methods

	void runAllWithMode(ExecutionMode mode);

	// Results and diagnostics

	void printConfiguration() const;
	void printSimulationSummary() const;
	
	// Getters
	const SimulationConfig& getConfig() const { return config_; }
	size_t getSimulationCount() const { return simulations_.size(); }
	const SimulationInstance& getSimulation(int instanceId) const;

	validateConfiguration();

void PricingOrchestrator::validateConfiguration() const
{
	if (!config_.isValid()) {
		throw std::invalid_argument("Invalid simulation configuration");
	}
	
	if (config_.mediatorType == MediatorType::MonteCarlo) {
		if (config_.sdeTypes.empty() || config_.fdmTypes.empty() || config_.rngTypes.empty()) {
			throw std::invalid_argument("Monte Carlo mediator requires SDE, FDM, and RNG types");
		}
	}
}

void PricingOrchestrator::initializeComponents()
{
	std::cout << "=== Initializing Pricing Orchestrator ===" << std::endl;
	std::cout << "Mediator Type: " << static_cast<int>(config_.mediatorType) << std::endl;
	std::cout << "Execution Mode: " << static_cast<int>(config_.executionMode) << std::endl;
	std::cout << "Number of Simulations per Mediator: " << config_.numSimulations << std::endl;
	
	// Store options
	options_.clear();
	for (const auto& opt : config_.options) {
		std::cout << "  - Option: " << opt.OptionName << std::endl;
		options_.push_back(opt);
	}
	
	// Calculate total combinations
	if (config_.mediatorType == MediatorType::MonteCarlo) {
		size_t totalCombinations = config_.sdeTypes.size() * 
		                          config_.fdmTypes.size() * 
		                          config_.rngTypes.size();
		std::cout << "\nTotal mediator combinations to create: " << totalCombinations << std::endl;
		std::cout << "  SDEs: " << config_.sdeTypes.size() << std::endl;
		std::cout << "  FDMs: " << config_.fdmTypes.size() << std::endl;
		std::cout << "  RNGs: " << config_.rngTypes.size() << std::endl;
	}
}

void PricingOrchestrator::buildAllMediators()
{
	std::cout << "\n=== Building All Mediators ===" << std::endl;
	
	simulations_.clear();
	int instanceId = 0;
	
	if (config_.mediatorType == MediatorType::MonteCarlo) {
		// Create a mediator for every combination of SDE, FDM, and RNG
		for (const auto& sdeType : config_.sdeTypes) {
			for (const auto& fdmType : config_.fdmTypes) {
				for (const auto& rngType : config_.rngTypes) {
					std::cout << "\nCreating simulation instance " << instanceId << ":" << std::endl;
					std::cout << "  SDE: " << sdeType << ", FDM: " << fdmType << ", RNG: " << rngType << std::endl;
					
					// Create simulation instance
					SimulationInstance instance(instanceId, sdeType, fdmType, rngType);
					
					// Use first option for component creation
					const auto& firstOption = config_.options[0];
					
					// Create components
					auto sde = createSde(sdeType, firstOption);
					auto fdm = createFdm(fdmType, sde, 500); // TODO: Get from config
					auto rng = createRng(rngType);
					
					if (!sde || !fdm || !rng) {
						std::cerr << "  Warning: Failed to create components for instance " << instanceId << std::endl;
						continue;
					}
					
					// Create mediator with these components
					auto parts = std::make_tuple(sde, fdm, rng);
					instance.mediator = std::make_unique<MonteCarloMediator>(parts, config_.numSimulations);
					
					std::cout << "  Successfully created mediator instance " << instanceId << std::endl;
					
					simulations_.push_back(std::move(instance));
					instanceId++;
				}
			}
		}
	}
	else {
		// For other mediator types (PDE, BinomialTree), create single instance
		SimulationInstance instance(instanceId, "", "", "");
		instance.mediator = MediatorFactory::createMediator(config_.mediatorType);
		instance.mediator->configure(config_);
		simulations_.push_back(std::move(instance));
	}
	
	std::cout << "\nTotal simulation instances created: " << simulations_.size() << std::endl;
}

	if (sdeType == "GBM") {
		return std::make_shared<GBM>(option.r, option.sig, option.D, option.IC, option.T);
	}
	else if (sdeType == "CEV") {
		return std::make_shared<CEV>(option.r, option.sig, option.D, option.IC, option.T, option.b);
	}
	// Add more SDE types as needed
	
	std::cerr << "Unknown SDE type: " << sdeType << std::endl;
	return nullptr;
}

}

void PricingOrchestrator::setupPricers()
{
	std::cout << "\n=== Setting Up Pricers ===" << std::endl;
	
	// Create pricers for each simulation instance
	for (auto& simulation : simulations_) {
		std::cout << "\nSetting up pricers for simulation instance " << simulation.instanceId << std::endl;
		
		// Create pricers for each (option, pricerType) combination
		for (const auto& option : config_.options) {
			for (const auto& pricerType : config_.pricerTypes) {
				auto pricer = PricerFactory::createPricer(pricerType, option);
				
				if (!pricer) {
					std::cerr << "  Warning: Failed to create pricer: " << pricerType << std::endl;
					continue;
				}
				
				simulation.pricers.push_back(pricer);
				std::cout << "  - Created " << pricerType << " for " << option.OptionName << std::endl;
			}
		}
		
		std::cout << "  Total pricers for instance " << simulation.instanceId << ": " 
		          << simulation.pricers.size() << std::endl;
	}
}

void PricingOrchestrator::connectPricers()
{
	std::cout << "\n=== Connecting Pricers ===" << std::endl;
	
	for (auto& simulation : simulations_) {
		if (!simulation.mediator) {
			std::cerr << "Warning: Simulation instance " << simulation.instanceId 
			          << " has no mediator" << std::endl;
			continue;
		}
		
		if (simulation.pricers.empty()) {
			std::cerr << "Warning: Simulation instance " << simulation.instanceId 
			          << " has no pricers" << std::endl;
			continue;
		}
		
		std::cout << "Connecting pricers for simulation instance " << simulation.instanceId << std::endl;
		
		int slot = 0;
		for (const auto& pricer : simulation.pricers) {
			simulation.mediator->connectPricer(pricer, slot);
			slot++;
		}
		
		std::cout << "  Connected " << slot << " pricers" << std::endl;
	}
}

void PricingOrchestrator::runAll()
{
	std::cout << "\n=== Running All Simulations ===" << std::endl;
	
	auto overallStart = std::chrono::high_resolution_clock::now();
	
	for (auto& simulation : simulations_) {
		std::cout << "\n--- Running Simulation Instance " << simulation.instanceId << " ---" << std::endl;
		std::cout << "Configuration: SDE=" << simulation.sdeType 
		          << ", FDM=" << simulation.fdmType 
		          << ", RNG=" << simulation.rngType << std::endl;
		
		auto start = std::chrono::high_resolution_clock::now();
		
		// Run based on execution mode
		switch (config_.executionMode) {
			case ExecutionMode::SingleThreaded:
				simulation.mediator->start();
				break;
			case ExecutionMode::OpenMP:
				simulation.mediator->startOpenMP();
				break;
			case ExecutionMode::PPL:
				simulation.mediator->startPPL();
				break;
		}
		
		auto end = std::chrono::high_resolution_clock::now();
		std::chrono::duration<double> elapsed = end - start;
		
		std::cout << "Instance " << simulation.instanceId << " completed in " 
		          << elapsed.count() << " seconds" << std::endl;
	}
	
	auto overallEnd = std::chrono::high_resolution_clock::now();
	std::chrono::duration<double> totalElapsed = overallEnd - overallStart;
	
	std::cout << "\n=== All Simulations Complete ===" << std::endl;
	std::cout << "Total execution time: " << totalElapsed.count() << " seconds" << std::endl;
	std::cout << "Average time per simulation: " 
	          << (totalElapsed.count() / simulations_.size()) << " seconds" << std::endl;
}

void PricingOrchestrator::runSimulation(int instanceId)
{
	if (instanceId < 0 || instanceId >= static_cast<int>(simulations_.size())) {
		throw std::out_of_range("Invalid simulation instance ID");
	}
	
	auto& simulation = simulations_[instanceId];
	
	std::cout << "\n--- Running Simulation Instance " << instanceId << " ---" << std::endl;
	std::cout << "Configuration: SDE=" << simulation.sdeType 
	          << ", FDM=" << simulation.fdmType 
	          << ", RNG=" << simulation.rngType << std::endl;
	
	auto start = std::chrono::high_resolution_clock::now();
	
	switch (config_.executionMode) {
		case ExecutionMode::SingleThreaded:
			simulation.mediator->start();
			break;
		case ExecutionMode::OpenMP:
			simulation.mediator->startOpenMP();
			break;
		case ExecutionMode::PPL:
			simulation.mediator->startPPL();
			break;
	}
	
	auto end = std::chrono::high_resolution_clock::now();
	std::chrono::duration<double> elapsed = end - start;
	
	std::cout << "Instance " << instanceId << " completed in " 
	          << elapsed.count() << " seconds" << std::endl;
}

================================================================================
KEY CONCEPTS
================================================================================

1. CARTESIAN PRODUCT
   - Creates all combinations: SDE₁×FDM₁×RNG₁, SDE₁×FDM₁×RNG₂, etc.
   - Each combination gets its own mediator instance
   - Enables comprehensive comparison of methods

2. SIMULATION INSTANCE
   - Container for one complete simulation setup
   - Holds: mediator, pricers, component types, ID
   - Can be run independently or as part of batch

3. COMPONENT FACTORIES
   - createSde(): Creates SDE from string type name
   - createFdm(): Creates FDM from string type name
   - createRng(): Creates RNG from string type name
   - Easily extensible for new component types

4. EXECUTION MODES
   - SingleThreaded: Sequential execution
   - OpenMP: Parallel using OpenMP
   - PPL: Parallel using Microsoft PPL

5. FLEXIBILITY
   - Run all simulations: runAll()
   - Run specific instance: runSimulation(id)
   - Override execution mode: runAllWithMode(mode)
   - Print diagnostics at any time
